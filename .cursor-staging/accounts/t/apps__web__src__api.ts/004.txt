export async function joinGame(name: string): Promise<PlayerView> {
  const res = await credFetch("/api/join", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ name }),
  });
  return parse<PlayerView>(res);
}
